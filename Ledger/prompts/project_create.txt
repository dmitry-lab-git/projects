CREATE TABLE development.projects (
	id uuid DEFAULT gen_random_uuid() NOT NULL,
	parent_id uuid NULL,
	code varchar(50) NOT NULL,
	"name" varchar(255) NOT NULL,
	entity_type varchar(50) NOT NULL,
	status varchar(50) DEFAULT 'active'::character varying NULL,
	started_at date NULL,
	completed_at date NULL,
	description text NULL,
	tags _text NULL,
	metadata jsonb DEFAULT '{}'::jsonb NULL,
	created_at timestamptz DEFAULT CURRENT_TIMESTAMP NULL,
	updated_at timestamptz DEFAULT CURRENT_TIMESTAMP NULL,
	CONSTRAINT projects_code_key UNIQUE (code),
	CONSTRAINT projects_code_not_null NOT NULL code,
	CONSTRAINT projects_entity_type_not_null NOT NULL entity_type,
	CONSTRAINT projects_id_not_null NOT NULL id,
	CONSTRAINT projects_name_not_null NOT NULL name,
	CONSTRAINT projects_pkey PRIMARY KEY (id),
	CONSTRAINT projects_parent_id_fkey FOREIGN KEY (parent_id) REFERENCES development.projects(id) ON DELETE SET NULL
);
CREATE INDEX idx_projects_parent_id ON development.projects USING btree (parent_id);
